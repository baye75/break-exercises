Array Utility Library

myMap(array, callback)

Creates a new array by calling a function on every element.

Example:

myMap([1,2,3], x => x * 2)

myFilter(array, callback)

Creates a new array with elements that pass a test.

Example:

myFilter([1,2,3,4], x => x % 2 === 0)

myReduce(array, callback, initial)

Reduces array to a single value (accumulator).

Example:

myReduce([1,2,3], (acc, x) => acc + x, 0)

Library Source Code (simplified)

// Custom map 
function myMap(arr, callback) { 
  const result = []; 
  for (let i = 0; i < arr.length; i++) { 
    result.push(callback(arr[i], i, arr)); 
  } 
  return result; 
} 
 
// Custom filter 
function myFilter(arr, callback) { 
  const result = []; 
  for (let i = 0; i < arr.length; i++) { 
    if (callback(arr[i], i, arr)) { 
      result.push(arr[i]); 
    } 
  } 
  return result; 
} 
 
// Custom reduce 
function myReduce(arr, callback, initial) { 
  let acc = initial !== undefined ? initial : arr[0]; 
  let start = initial !== undefined ? 0 : 1; 
  for (let i = start; i < arr.length; i++) { 
    acc = callback(acc, arr[i], i, arr); 
  } 
  return acc; 
} 
 
// Custom flat 
function myFlat(arr, depth = 1) { 
  const result = []; 
  for (let i = 0; i < arr.length; i++) { 
    if (Array.isArray(arr[i]) && depth > 0) { 
      const nested = myFlat(arr[i], depth - 1); 
      for (let j = 0; j < nested.length; j++) { 
        result.push(nested[j]); 
      } 
    } else { 
      result.push(arr[i]); 
    } 
  } 
  return result; 
} 
 
// Custom debounce 
function myDebounce(fn, delay) { 
  let timeoutId; 
  return function(...args) { 
    clearTimeout(timeoutId); 
    timeoutId = setTimeout(() => fn.apply(this, args), delay); 
  }; 
} 
 
// Custom throttle 
function myThrottle(fn, limit) { 
  let inThrottle = false; 
  return function(...args) { 
    if (!inThrottle) { 
      fn.apply(this, args); 
      inThrottle = true; 
      setTimeout(() => { inThrottle = false; }, limit); 
    } 
  }; 
}
